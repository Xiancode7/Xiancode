function Login-KeyAuth {
    param ($LicenseKey)
    
    Write-Host "[!] Connecting to Server..." -ForegroundColor Yellow
    
    # 1. INIT
    $initBody = @{
        type = "init"
        ver = $VERSION
        name = $APP_NAME
        ownerid = $OWNER_ID
        # secret = $APP_SECRET # Uncomment if secret is required by your KeyAuth app version
    }
    
    try {
        $initResponse = Invoke-RestMethod -Uri $KEYAUTH_API -Method Post -Body $initBody -ErrorAction Stop
        
        if ($initResponse.success -ne $true) {
            Write-Host "Server Init Failed: $($initResponse.message)" -ForegroundColor Red
            return $false
        }
        
        $sessionid = $initResponse.sessionid
        
        # 2. LICENSE LOGIN
        $hwid = (Get-CimInstance win32_computersystemproduct).UUID
        $loginBody = @{
            type = "license"
            key = $LicenseKey
            hwid = $hwid
            sessionid = $sessionid
            name = $APP_NAME
            ownerid = $OWNER_ID
        }
        
        $loginResponse = Invoke-RestMethod -Uri $KEYAUTH_API -Method Post -Body $loginBody -ErrorAction Stop
        
        if ($loginResponse.success -eq $true) {
            Write-Host "Login Success! Welcome, $($loginResponse.info.username)" -ForegroundColor Green
            return $true
        } else {
            Write-Host "Login Failed: $($loginResponse.message)" -ForegroundColor Red
            return $false
        }
    }
    catch {
        Write-Host "Connection Error: $_" -ForegroundColor Red
        return $false
    }
}
# ================= HELPER FUNCTIONS =================
function Get-RamInfo {
    $os = Get-CimInstance Win32_OperatingSystem
    $total = [Math]::Round($os.TotalVisibleMemorySize / 1MB, 2)
    $free = [Math]::Round($os.FreePhysicalMemory / 1MB, 2)
    $used = [Math]::Round($total - $free, 2)
    return @{Total=$total; Used=$used; Free=$free}
}
function Clean-Folder ($Path) {
    if (Test-Path $Path) {
        $files = Get-ChildItem -Path $Path -Recurse -File -ErrorAction SilentlyContinue
        $size = ($files | Measure-Object -Property Length -Sum).Sum
        $count = $files.Count
        Remove-Item -Path "$Path\*" -Recurse -Force -ErrorAction SilentlyContinue
        return @{Size=$size; Count=$count}
    }
    return @{Size=0; Count=0}
}
function Optimize-Network {
    Write-Host "   >>> Optimizing Network Stack..." -ForegroundColor Cyan
    # TCP Optimizations
    $tcpParams = @{
        "TcpAckFrequency" = 1
        "TCPNoDelay" = 1
        "TcpDelAckTicks" = 0
    }
    
    $interfaces = "HKLM\SYSTEM\CurrentControlSet\Services\Tcpip\Parameters\Interfaces"
    Get-ChildItem -Path "Registry::$interfaces" | ForEach-Object {
        $p = $_.Name.Replace("HKEY_LOCAL_MACHINE", "HKLM")
        foreach ($key in $tcpParams.Keys) {
            reg add "$p" /v "$key" /t REG_DWORD /d $tcpParams[$key] /f >$null 2>&1
        }
    }
    # Custom optimizations for lower latency
    netsh int tcp set global autotuninglevel=normal
    netsh int tcp set global rss=enabled
    netsh int tcp set global rsc=disabled
    netsh int tcp set global chimneymonitor=disabled
    netsh int tcp set global ecncapability=enabled
    netsh int tcp set global timestamps=disabled
    netsh int tcp set global nonsackrttresiliency=disabled
    netsh int tcp set global initialrto=2000
    netsh int tcp set global rtoversions=1
}
function Optimize-InputLag {
    Write-Host "   >>> Reducing Input Lag..." -ForegroundColor Cyan
    # Keyboard Queue Size
    reg add "HKLM\SYSTEM\CurrentControlSet\Services\kbdclass\Parameters" /v "KeyboardDataQueueSize" /t REG_DWORD /d 50 /f >$null 2>&1
    # Mouse Queue Size
    reg add "HKLM\SYSTEM\CurrentControlSet\Services\mouclass\Parameters" /v "MouseDataQueueSize" /t REG_DWORD /d 50 /f >$null 2>&1
    # Priority Control
    reg add "HKLM\SYSTEM\CurrentControlSet\Control\PriorityControl" /v "Win32PrioritySeparation" /t REG_DWORD /d 38 /f >$null 2>&1
    # Disable Mouse Acceleration
    reg add "HKCU\Control Panel\Mouse" /v "MouseSpeed" /t REG_SZ /d "0" /f >$null 2>&1
    reg add "HKCU\Control Panel\Mouse" /v "MouseThreshold1" /t REG_SZ /d "0" /f >$null 2>&1
    reg add "HKCU\Control Panel\Mouse" /v "MouseThreshold2" /t REG_SZ /d "0" /f >$null 2>&1
}
