chcp 65001 > $null
$OutputEncoding = [System.Text.Encoding]::UTF8

# ================= KEYAUTH CONFIG =================
function _d($x) { [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($x)) }
$APP_NAME = _d("WGlhbkVYVFJFTUU=") 
$OWNER_ID = _d("blpjZDN4MkVYOQ==") 
$VERSION  = "1.0"
$KEYAUTH_API = "https://keyauth.win/api/1.3/"

# ================= ADMIN CHECK =================
if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "ERROR: Please run PowerShell as Administrator." -ForegroundColor Red
    Pause ; Exit
}

# ================= GET HWID & INIT =================
function Get-HWID {
    $cpu = (Get-CimInstance Win32_Processor).ProcessorId
    $bios = (Get-CimInstance Win32_BIOS).SerialNumber
    $disk = (Get-CimInstance Win32_PhysicalMedia | Select-Object -First 1).SerialNumber
    return "$cpu|$bios|$disk"
}
$HWID = Get-HWID

Write-Host "Connecting to Server..." -ForegroundColor Gray
try {
    $init_res = Invoke-RestMethod -Uri "$($KEYAUTH_API)?type=init&name=$APP_NAME&ownerid=$OWNER_ID&version=$VERSION" -Method GET
    if ($init_res.success -ne $true) { Write-Host "Init Failed." -ForegroundColor Red ; Pause ; Exit }
    $sessionid = $init_res.sessionid
} catch { Write-Host "Connection Error." -ForegroundColor Red ; Pause ; Exit }

# ================= LOGIN =================
Clear-Host
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "     FiveM Boost FPS PRO EXTREME V.8" -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan
$license = Read-Host "Enter License Key"

try {
    $res = Invoke-RestMethod -Uri "$($KEYAUTH_API)?type=license&key=$license&hwid=$HWID&sessionid=$sessionid&name=$APP_NAME&ownerid=$OWNER_ID" -Method GET
    if ($res.success -ne $true) { Write-Host "ERROR: $($res.message)" -ForegroundColor Red ; Pause ; Exit }
} catch { Write-Host "Login Failed." -ForegroundColor Red ; Pause ; Exit }

# ================= MENU =================
do {
    Clear-Host
    Write-Host "==========================================" -ForegroundColor Cyan
    Write-Host "    XIANCODE MANUTAU Setting " -ForegroundColor Cyan
    Write-Host "==========================================" -ForegroundColor Cyan
    Write-Host "[1] ENABLE ALL TWEAKS (FORCE STATUS TRUE)"
    Write-Host "[2] RESTORE SYSTEM DEFAULTS"
    Write-Host "[3] SHOW STATUS (CHECK TRUE/FALSE)"
    Write-Host "[4] CHECK CPU TEMPERATURE"
    Write-Host "[5] CLEAN CACHE ONLY (TEMP/FIVEM)"
    Write-Host "[0] EXIT"
    Write-Host ""
    $choice = Read-Host "Select option"

    switch ($choice) {
        "1" {
            Write-Host "`n[!] Activating MMAgent & Core Performance..." -ForegroundColor Yellow
            
            # --- 1. บังคับเปิด SysMain เพื่อให้ MMAgent เป็น True ---
            Set-Service SysMain -StartupType Automatic -ErrorAction SilentlyContinue
            Start-Service SysMain -ErrorAction SilentlyContinue
            
            # --- 2. ตั้งค่า MMAgent เป็น True ทั้งหมด ---
            Enable-MMAgent -ApplicationLaunchPrefetching -ApplicationPreLaunch -MemoryCompression -PageCombining
            
            # --- 3. บังคับค่า MaxOperationAPIFiles เป็น 512 ---
            $apiPath = "HKLM\Software\Microsoft\Windows\CurrentVersion\WINEVT\Channels\Microsoft-Windows-Store/Operational"
            if (!(Test-Path $apiPath)) { New-Item -Path $apiPath -Force >$null }
            reg add "$apiPath" /v "MaxOperationAPIFiles" /t REG_DWORD /d 512 /f >$null

            # --- 4. Network & Latency (Ping) ---
            reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Multimedia\SystemProfile" /v "NetworkThrottlingIndex" /t REG_DWORD /d 0xFFFFFFFF /f >$null
            $interfaces = "HKLM\SYSTEM\CurrentControlSet\Services\Tcpip\Parameters\Interfaces"
            Get-ChildItem -Path "Registry::$interfaces" | ForEach-Object {
                $p = $_.Name.Replace("HKEY_LOCAL_MACHINE", "HKLM")
                reg add "$p" /v "TcpAckFrequency" /t REG_DWORD /d 1 /f >$null
                reg add "$p" /v "TCPNoDelay" /t REG_DWORD /d 1 /f >$null
                reg add "$p" /v "TcpDelAckTicks" /t REG_DWORD /d 0 /f >$null
            }

            # --- 5. CPU & GPU Priority ---
            powercfg -setacvalueindex scheme_current sub_processor cpmincores 100
            powercfg -setactive scheme_current
            reg add "HKLM\SYSTEM\CurrentControlSet\Control\PriorityControl" /v "Win32PrioritySeparation" /t REG_DWORD /d 38 /f >$null
            reg add "HKLM\SYSTEM\CurrentControlSet\Control\GraphicsDrivers" /v "HwSchMode" /t REG_DWORD /d 2 /f >$null

            # --- 6. Disable Throttling & Update ---
            Stop-Service wuauserv -Force -ErrorAction SilentlyContinue
            Set-Service wuauserv -StartupType Disabled
            Set-MpPreference -DisableRealtimeMonitoring $true -ErrorAction SilentlyContinue

            # --- 7. Cleaner ---
            $CleanPaths = @("$env:TEMP\*", "$env:SystemRoot\Temp\*", "$env:SystemRoot\Prefetch\*")
            foreach ($path in $CleanPaths) { Remove-Item $path -Recurse -Force -ErrorAction SilentlyContinue }

            Write-Host "ULTIMATE V.8 SUCCESS: All values set to TRUE!" -ForegroundColor Green
            Pause
        }

        "2" {
            Write-Host "`n[!] Restoring System Defaults..." -ForegroundColor Yellow
            Disable-MMAgent -ApplicationLaunchPrefetching -ApplicationPreLaunch
            Set-Service wuauserv -StartupType Automatic -ErrorAction SilentlyContinue
            powercfg -setactive SCHEME_BALANCED
            Write-Host "Restore Completed." -ForegroundColor Green
            Pause
        }

        "3" {
            Write-Host "`n--- Current System Status ---" -ForegroundColor Cyan
            $agent = Get-MMAgent
            $regValue = (Get-ItemProperty "HKLM\Software\Microsoft\Windows\CurrentVersion\WINEVT\Channels\Microsoft-Windows-Store/Operational" -Name MaxOperationAPIFiles -ErrorAction SilentlyContinue).MaxOperationAPIFiles
            
            Write-Host "ApplicationLaunchPrefetching : $($agent.ApplicationLaunchPrefetching)"
            Write-Host "ApplicationPreLaunch         : $($agent.ApplicationPreLaunch)"
            Write-Host "MaxOperationAPIFiles         : $(if($regValue){$regValue}else{"Not Set"})"
            Write-Host "MemoryCompression            : $($agent.MemoryCompression)"
            Write-Host "PageCombining                : $($agent.PageCombining)"
            Write-Host "PSComputerName               : $($agent.PSComputerName)"
            Pause
        }

        "4" {
            Write-Host "`n[!] Checking CPU Temperature..." -ForegroundColor Yellow
            try {
                $tempK = (Get-CimInstance -Namespace root/WMI -ClassName MSAcpi_ThermalZoneTemperature).CurrentTemperature
                $tempC = ($tempK / 10) - 273.15
                Write-Host "Current CPU Temperature: $([Math]::Round($tempC, 2)) °C" -ForegroundColor Cyan
            } catch { Write-Host "Hardware not supported." -ForegroundColor Red }
            Pause
        }

        "5" {
            Remove-Item "$env:TEMP\*" -Recurse -Force -ErrorAction SilentlyContinue
            Write-Host "Cache Cleaned." -ForegroundColor Green
            Pause
        }

        "0" { Exit }
    }
} while ($choice -ne "0")
