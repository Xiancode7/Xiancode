# ---- ตั้งค่าการแสดงผลภาษาไทย ----
chcp 65001 > $null
$OutputEncoding = [System.Text.Encoding]::UTF8

$SHOP_NAME = "Xiancode"

# ================= KEYAUTH CONFIG =================
function _d($x) {
    [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($x))
}

# ข้อมูลแอปของคุณ (เข้ารหัสไว้เพื่อความปลอดภัยเบื้องต้น)
$APP_NAME = _d("WGlhbkVYVFJFTUU=") # XianEXTREME
$OWNER_ID = _d("blpjZDN4MkVYOQ==") # nZcd3x2EX9
$VERSION  = "1.0"
$KEYAUTH_API = "https://keyauth.win/api/1.3/"

# ================= ADMIN CHECK =================
if (-not ([Security.Principal.WindowsPrincipal] `
    [Security.Principal.WindowsIdentity]::GetCurrent()
).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "ERROR: Please run PowerShell as Administrator." -ForegroundColor Red
    Pause ; Exit
}

# ================= GET HWID =================
function Get-HWID {
    $cpu  = (Get-CimInstance Win32_Processor).ProcessorId
    $bios = (Get-CimInstance Win32_BIOS).SerialNumber
    $disk = (Get-CimInstance Win32_PhysicalMedia | Select-Object -First 1).SerialNumber
    return "$cpu|$bios|$disk"
}
$HWID = Get-HWID

# ================= [สำคัญ] STEP 1: INITIALIZE SESSION =================
# จุดนี้คือการแก้ Error "Session ID not provided" โดยตรง
Clear-Host
Write-Host "Connecting to KeyAuth Server..." -ForegroundColor Gray

$init_url = "$($KEYAUTH_API)?type=init&name=$APP_NAME&ownerid=$OWNER_ID&version=$VERSION"

try {
    $init_res = Invoke-RestMethod -Uri $init_url -Method GET
    if ($init_res.success -ne $true) {
        Write-Host "ERROR: Init Failed - $($init_res.message)" -ForegroundColor Red
        Pause ; Exit
    }
    # ดึง Session ID ที่เซิร์ฟเวอร์เพิ่งสร้างให้มาเก็บไว้
    $sessionid = $init_res.sessionid
} catch {
    Write-Host "ERROR: Cannot connect to KeyAuth API. Please check internet." -ForegroundColor Red
    Pause ; Exit
}

# ================= STEP 2: LOGIN SYSTEM =================
Clear-Host
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "     FiveM Boost FPS PRO EXTREME" -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host " Shop: $SHOP_NAME"
Write-Host ""

$license = Read-Host "Enter License Key"

# 👉 ส่ง Request พร้อม Session ID ที่ได้จาก Step 1
$login_url = "$($KEYAUTH_API)?type=license&key=$license&hwid=$HWID&sessionid=$sessionid&name=$APP_NAME&ownerid=$OWNER_ID"

try {
    $res = Invoke-RestMethod -Uri $login_url -Method GET
    
    if ($res.success -ne $true) {
        Write-Host "ERROR: $($res.message)" -ForegroundColor Red
        Pause ; Exit
    }
} catch {
    Write-Host "ERROR: Login request failed." -ForegroundColor Red
    Pause ; Exit
}

Write-Host "License verified successfully! Welcome." -ForegroundColor Green
Start-Sleep -Seconds 1

# ================= MENU =================
do {
    Clear-Host
    Write-Host "==========================================" -ForegroundColor Cyan
    Write-Host "     FiveM Boost FPS PRO - EXTREME" -ForegroundColor Cyan
    Write-Host "==========================================" -ForegroundColor Cyan
    Write-Host "[1] BOOST FPS (EXTREME MODE)"
    Write-Host "[2] RESTORE DEFAULT SETTINGS"
    Write-Host "[3] SHOW MEMORY STATUS (MMAgent)"
    Write-Host "[0] EXIT"
    Write-Host ""

    $choice = Read-Host "Select option"

    switch ($choice) {
        "1" {
            Write-Host "`nEnabling EXTREME performance mode..." -ForegroundColor Yellow
            Disable-MMAgent -MemoryCompression
            Stop-Service SysMain -Force -ErrorAction SilentlyContinue
            Set-Service SysMain -StartupType Disabled
            powercfg -setactive SCHEME_MIN
            
            # Registry Tweaks
            reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Multimedia\SystemProfile" /v SystemResponsiveness /t REG_DWORD /d 0 /f >$null
            reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Multimedia\SystemProfile\Tasks\Games" /v "Scheduling Category" /t REG_SZ /d High /f >$null
            reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Multimedia\SystemProfile\Tasks\Games" /v "Priority" /t REG_DWORD /d 6 /f >$null
            reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Multimedia\SystemProfile\Tasks\Games" /v "GPU Priority" /t REG_DWORD /d 8 /f >$null
            reg add "HKLM\SYSTEM\CurrentControlSet\Control\Power\PowerThrottling" /v PowerThrottlingOff /t REG_DWORD /d 1 /f >$null
            reg add "HKLM\SYSTEM\CurrentControlSet\Control\Session Manager\Memory Management" /v DisablePagingExecutive /t REG_DWORD /d 1 /f >$null
            
            # Mouse & Game Mode
            reg add "HKCU\Control Panel\Mouse" /v MouseSpeed /t REG_SZ /d 0 /f >$null
            reg add "HKCU\System\GameConfigStore" /v GameDVR_Enabled /t REG_DWORD /d 0 /f >$null
            
            foreach ($s in "DiagTrack","MapsBroker","lfsvc","SharedAccess") {
                Stop-Service $s -Force -ErrorAction SilentlyContinue
                Set-Service $s -StartupType Disabled -ErrorAction SilentlyContinue
            }
            Write-Host "EXTREME mode enabled. Please restart your PC." -ForegroundColor Green
            Pause
        }
        "2" {
            Write-Host "`nRestoring default system settings..." -ForegroundColor Yellow
            Enable-MMAgent -MemoryCompression
            Set-Service SysMain -StartupType Automatic
            Start-Service SysMain -ErrorAction SilentlyContinue
            powercfg -setactive SCHEME_BALANCED
            reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Multimedia\SystemProfile" /v SystemResponsiveness /t REG_DWORD /d 20 /f >$null
            Write-Host "System settings restored." -ForegroundColor Green
            Pause
        }
        "3" { Get-MMAgent | Format-List; Pause }
        "0" { Exit }
    }
} while ($choice -ne "0")
