chcp 65001 > $null
$OutputEncoding = [System.Text.Encoding]::UTF8

# ================= KEYAUTH CONFIG =================
function _d($x) { [Text.Encoding]::UTF8.GetString([Convert]::FromBase64String($x)) }
$APP_NAME = _d("WGlhbkVYVFJFTUU=") 
$OWNER_ID = _d("blpjZDN4MkVYOQ==") 
$VERSION  = "1.0"
$KEYAUTH_API = "https://keyauth.win/api/1.3/"

# ================= ADMIN CHECK =================
if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "ERROR: Please run PowerShell as Administrator." -ForegroundColor Red
    Pause ; Exit
}

# ================= GET HWID & INIT =================
function Get-HWID {
    $cpu = (Get-CimInstance Win32_Processor).ProcessorId
    $bios = (Get-CimInstance Win32_BIOS).SerialNumber
    $disk = (Get-CimInstance Win32_PhysicalMedia | Select-Object -First 1).SerialNumber
    return "$cpu|$bios|$disk"
}
$HWID = Get-HWID

Write-Host "Connecting to Server..." -ForegroundColor Gray
try {
    $init_res = Invoke-RestMethod -Uri "$($KEYAUTH_API)?type=init&name=$APP_NAME&ownerid=$OWNER_ID&version=$VERSION" -Method GET
    if ($init_res.success -ne $true) { Write-Host "Init Failed." -ForegroundColor Red ; Pause ; Exit }
    $sessionid = $init_res.sessionid
} catch { Write-Host "Connection Error." -ForegroundColor Red ; Pause ; Exit }

# ================= LOGIN =================
Clear-Host
Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "      XIANEXTREME V.1" -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan
$license = Read-Host "Enter License Key"

try {
    $res = Invoke-RestMethod -Uri "$($KEYAUTH_API)?type=license&key=$license&hwid=$HWID&sessionid=$sessionid&name=$APP_NAME&ownerid=$OWNER_ID" -Method GET
    if ($res.success -ne $true) { Write-Host "ERROR: $($res.message)" -ForegroundColor Red ; Pause ; Exit }
} catch { Write-Host "Login Failed." -ForegroundColor Red ; Pause ; Exit }

# ================= MENU =================
do {
    Clear-Host
    Write-Host "==========================================" -ForegroundColor Cyan
    Write-Host "   XIANCODE MANUTAU SETTING " -ForegroundColor Cyan
    Write-Host "==========================================" -ForegroundColor Cyan
    Write-Host "[1] ENABLE ALL TWEAKS (SET TO TRUE/512)"
    Write-Host "[2] RESTORE SYSTEM DEFAULTS (ALL)"
    Write-Host "[3] SHOW STATUS (CHECK VALUES)"
    Write-Host "[4] CHECK CPU TEMPERATURE"
    Write-Host "[5] CLEAN CACHE (TEMP/FIVEM)"
    Write-Host "[6] DISABLE MMAGENT TWEAKS (SET TO FALSE)"
    Write-Host "[0] EXIT"
    Write-Host ""
    $choice = Read-Host "Select option"

    switch ($choice) {
        "1" {
            Write-Host "`n[!] Activating MMAgent & Force True..." -ForegroundColor Yellow
            Set-Service SysMain -StartupType Automatic -ErrorAction SilentlyContinue
            Start-Service SysMain -ErrorAction SilentlyContinue
            Enable-MMAgent -ApplicationLaunchPrefetching -ApplicationPreLaunch -MemoryCompression -PageCombining
            
            $regPath = "HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\WINEVT\Channels\Microsoft-Windows-Store/Operational"
            if (!(Test-Path $regPath)) { New-Item -Path $regPath -Force >$null }
            reg add "$regPath" /v "MaxOperationAPIFiles" /t REG_DWORD /d 512 /f >$null
            
            Write-Host "SUCCESS: Values set to TRUE / 512" -ForegroundColor Green
            Pause
        }

        "2" {
            Write-Host "`n[!] Restoring All Defaults..." -ForegroundColor Yellow
            Disable-MMAgent -ApplicationLaunchPrefetching -ApplicationPreLaunch
            Set-Service wuauserv -StartupType Automatic -ErrorAction SilentlyContinue
            powercfg -setactive SCHEME_BALANCED
            Write-Host "Restore Completed." -ForegroundColor Green
            Pause
        }

        "3" {
            Write-Host "`n--- Current System Status ---" -ForegroundColor Cyan
            $agent = Get-MMAgent
            $regVal = (Get-ItemProperty "HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\WINEVT\Channels\Microsoft-Windows-Store/Operational" -Name MaxOperationAPIFiles -ErrorAction SilentlyContinue).MaxOperationAPIFiles
            
            Write-Host "ApplicationLaunchPrefetching : $($agent.ApplicationLaunchPrefetching)"
            Write-Host "ApplicationPreLaunch         : $($agent.ApplicationPreLaunch)"
            Write-Host "MaxOperationAPIFiles         : $(if($regVal){$regVal}else{"Not Set"})"
            Write-Host "MemoryCompression            : $($agent.MemoryCompression)"
            Write-Host "PageCombining                : $($agent.PageCombining)"
            Write-Host "PSComputerName               : $($agent.PSComputerName)"
            Pause
        }

        "4" {
            try {
                $tempK = (Get-CimInstance -Namespace root/WMI -ClassName MSAcpi_ThermalZoneTemperature).CurrentTemperature
                Write-Host "CPU Temperature: $(($tempK/10)-273.15) °C" -ForegroundColor Cyan
            } catch { Write-Host "Not supported." -ForegroundColor Red }
            Pause
        }

        "5" {
            Remove-Item "$env:TEMP\*" -Recurse -Force -ErrorAction SilentlyContinue
            Write-Host "Cleaned." -ForegroundColor Green
            Pause
        }

        "6" {
            Write-Host "`n[!] Disabling MMAgent Tweaks (Set to False)..." -ForegroundColor Yellow
            
            # ปิด MMAgent
            Disable-MMAgent -ApplicationLaunchPrefetching -ApplicationPreLaunch -MemoryCompression -PageCombining
            
            # ลบค่า MaxOperationAPIFiles หรือตั้งกลับเป็นค่าเริ่มต้น
            $regPath = "HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\WINEVT\Channels\Microsoft-Windows-Store/Operational"
            reg delete "$regPath" /v "MaxOperationAPIFiles" /f >$null 2>&1
            
            Write-Host "SUCCESS: Values set to FALSE / Removed" -ForegroundColor Green
            Pause
        }

        "0" { Exit }
    }
} while ($choice -ne "0")
